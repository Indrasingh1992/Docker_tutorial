#!/bin/bash
set -e  # Exit immediately if a command exits with a non-zero status

echo "🔄 Updating package list..."
sudo apt-get update -y

echo "🐳 Installing Docker..."
sudo apt-get install -y docker.io

echo "👤 Adding user '$USER' to the 'docker' group..."
sudo usermod -aG docker $USER

echo "⚠️ 'newgrp docker' will start a new shell — the script will stop here if used directly."
echo "🟡 Please log out and log back in after this script to apply group change."
# Remove or comment this line if you want to avoid shell transition
newgrp docker
